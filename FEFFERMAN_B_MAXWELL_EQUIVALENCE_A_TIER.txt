1. EXACT ROOT PROPOSITION

  NSEMaxwell.Classical.ExactEquivalence
  := forall nu : Real, 0 < nu -> forall c : Real, 0 < c ->
       (FeffermanB nu <-> MaxwellB nu c).

  FeffermanB nu
  := forall a : Field, Data a ->
       exists u : Velocity, exists p : Pressure, NSSolution nu a u p.

  MaxwellB nu c
  := forall m : Vec, forall B0 : Field, MagneticData B0 ->
       exists B E : Velocity, exists rho : Pressure, exists J : Velocity,
         MaxwellSolution nu c m B0 B E rho J.

2. CONCRETE DOMAIN AND OPERATORS

  Vec = EuclideanSpace Real (Fin 3).
  Field = Vec -> Vec.
  Velocity = Real -> Field.
  Pressure = Real -> Vec -> Real.
  Spatial domain = unit-periodic lifts on Vec.
  Time domain = [0,infinity), including t = 0.
  Smoothness = joint ContDiffOn Real infinity on forward spacetime.
  Initial smoothness = ContDiff Real infinity.
  Spatial derivatives = Mathlib fderiv.
  Time derivatives = Mathlib derivWithin on Ici 0.
  Spatial means = Lebesgue integrals over the unit cube.
  Velocity mean = arbitrary m in Vec.
  Pressure = periodic; spatial-constant time-dependent gauge unrestricted.
  Forcing = zero.

  Lf(x) = sum_{k in Z^3} lambda(k)
            integral_[0,1]^3 cos(2*pi*k dot (x-y)) f(y) dy,
  lambda(0)=0,
  lambda(k)=(4*pi^2*|k|^2)^(-1), k != 0.
  K = curl L.
  U_m[B] = m + K B.
  G_nu,m[B] = B cross U_m[B] + nu curl B.
  V_nu,m[B] = -curl G_nu,m[B].
  Q_nu,m[B] = div G_nu,m[B].
  DG_nu,m[B](h) = h cross U_m[B] + B cross K h + nu curl h.
  J_nu,c,m[B] = c^2 curl B - DG_nu,m[B](V_nu,m[B]).
  pB = L Q[B] - |U_m[B]|^2/2 + mean(|U_m[B]|^2)/2.

  The Fourier expressions are total definitions.
  Their analytic properties are not consequences of their declarations alone.
