  ∃ C : ℝ, 0 ≤ C ∧ ∀ N : ℕ, 1 ≤ N →

NSSolution ν a u p =
  velocity_smooth : SmoothVelocity u
  pressure_smooth : SmoothPressure p
  velocity_periodic : PeriodicVelocity u
  pressure_periodic : PeriodicPressure p
  divergence_free : ∀ t, 0 ≤ t → ∀ x, divergence (u t) x = 0
  equation : ∀ t, 0 ≤ t → ∀ x,
    timeDerivative u t x + convection (u t) x + gradient (p t) x =
      ν • laplacian (u t) x
  initial : u 0 = a

forwardSpacetime = {z : ℝ × Vec | 0 ≤ z.1}
SmoothVelocity u = ContDiffOn ℝ ∞ (fun z => u z.1 z.2) forwardSpacetime
SmoothPressure p = ContDiffOn ℝ ∞ (fun z => p z.1 z.2) forwardSpacetime
timeDerivative u t x = derivWithin (fun s => u s x) (Ici 0) t
