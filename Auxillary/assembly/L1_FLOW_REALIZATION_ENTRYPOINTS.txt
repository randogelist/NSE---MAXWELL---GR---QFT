I := Set.Icc (0 : R) T; S := I x EuclideanSpace R (Fin 3).

L1:
  T>0;
  SmoothVelocityOn T u;
  forall t in I, SpatialPeriodic (u t);

  -> exists eta rho,
       MaterialPathOn T eta rho
