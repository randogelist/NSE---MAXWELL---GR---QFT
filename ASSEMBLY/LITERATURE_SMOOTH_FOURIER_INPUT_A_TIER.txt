  forall f : EuclideanSpace R (Fin n) -> C,
  ContDiff R infinity f -> LiteraturePeriodic f ->
  forall w : List (Fin n),
    (forall k,
      FourierCoefficient(PartialWord(w,f),k)
      = FourierWordMultiplier(w,k) * FourierCoefficient(f,k))
    and Summable(k |-> norm(FourierWordMultiplier(w,k)*FourierCoefficient(f,k)))
    and HasSumUniformly
      (k,x |-> FourierWordMultiplier(w,k)*FourierCoefficient(f,k)*Character(k,x))
      (PartialWord(w,f)).

  Character(k,x) = exp(2*pi*i*sum_j k_j*x_j).
  Uniformity domain = entire EuclideanSpace R (Fin n).
  Summation filter = finite frequency subsets ordered by inclusion.
  Domain hypotheses = smoothness and unit periodicity.
